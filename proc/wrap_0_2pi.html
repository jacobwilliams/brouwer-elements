    pure subroutine wrap_0_2pi(angle)
        !! Wraps an angle in radians to the range [0, 2*pi).
        real(wp), intent(inout) :: angle
        angle = modulo(angle, two_pi)
        if (angle < 0.0_wp) angle = angle + two_pi
    end subroutine wrap_0_2pi
