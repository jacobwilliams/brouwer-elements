    pure subroutine wrap_0_360(angle)
        !! Wraps an angle in degrees to the range [0, 360).
        real(wp), intent(inout) :: angle
        angle = modulo(angle, 360.0_wp)
        if (angle < 0.0_wp) angle = angle + 360.0_wp
    end subroutine wrap_0_360
